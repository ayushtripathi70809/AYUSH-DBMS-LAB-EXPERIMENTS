-- Create Table
CREATE TABLE Sales (
    SaleID INT PRIMARY KEY,
    CustomerID INT,
    ProductID INT,
    Amount DECIMAL(10,2),
    SaleDate DATE,
    City VARCHAR(50)
);

-- Generate 10000 Records
DELIMITER //

CREATE PROCEDURE GenerateData()
BEGIN
    DECLARE i INT DEFAULT 1;

    WHILE i <= 10000 DO
        INSERT INTO Sales VALUES (
            i,
            i % 1000,
            i % 100,
            (i % 5000) + 100,
            DATE_ADD('2025-01-01',
                INTERVAL (i % 365) DAY),
            CASE
                WHEN i % 3 = 0 THEN 'Delhi'
                WHEN i % 3 = 1 THEN 'Mumbai'
                ELSE 'Kanpur'
            END
        );

        SET i = i + 1;
    END WHILE;
END //

DELIMITER ;

CALL GenerateData();

-- Verify Number of Records
SELECT COUNT(*) FROM Sales;

-- QUERY 5: Search by Customer and Date
-- Before Optimization
EXPLAIN

WHERE CustomerID = 105

CREATE INDEX idx_customer_date
ON Sales(CustomerID, SaleDate);

-- Query Rewriting Example
-- Original Query
SELECT * FROM Sales
WHERE YEAR(SaleDate) = 2025;

-- Optimized Query
SELECT * FROM Sales
WHERE SaleDate >= '2025-01-01'
AND SaleDate < '2026-01-01';
